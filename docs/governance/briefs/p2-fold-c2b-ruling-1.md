```sql
CREATE FUNCTION public.listing_caps(p_listing uuid, p_country character DEFAULT NULL::bpchar)
RETURNS jsonb LANGUAGE sql STABLE SET search_path TO 'public' AS $fn$

           'plan',          r.key,

$fn$;
```
