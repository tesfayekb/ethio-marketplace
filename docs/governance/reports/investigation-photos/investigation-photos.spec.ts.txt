import { writeFileSync } from "node:fs";

import type { Page } from "@playwright/test";
import { PNG } from "pngjs";

import { test } from "./fixtures";
import { adPageFixtures } from "./helpers/ad-page";
import { BUCKET, madeJpeg } from "./helpers/photos";
import { RUN, rand } from "./helpers/posting";
import { gotoReady, settled, signInViaSession } from "./helpers/ui";
import { adminClient } from "./helpers/users";

/**
 * INVESTIGATION — investigation-photos, ruling 1 item 1: a MEASURING spec. It asserts nothing.
 *
 * The operator's walk of 2026-11-12 saw the ad photos on My listings and the ad pages drawn
 * malformed at large widths, and the category pictures blank. This spec draws those screens at
 * 1280 and 1024 (desktop-1280) and 360 (mobile-360) for one scratch seller with three ads — a
 * landscape photo, a portrait photo, and no photo under a category that has a picture (the
 * category fallback) — and records, for every `img` on each screen, what the ruling names: the
 * natural size, the drawn box, `object-fit`, the frame's width and inline style, the nearest
 * `[data-picture]` box's width and `container-type`, every ancestor that declares a
 * `container-type`, and the network answer (status, content type, bytes). One screenshot per
 * screen and width. The same file runs unchanged at 2b11a9d29, db13e13b7 and 6457836f3.
 *
 * The stored photos have the shape the browser's encoder writes (photo-encode.ts EDGES: cover
 * 1600, mid 960, card 480, thumb 160 on the long edge, the ratio kept); the category picture has
 * the admin pipeline's card size (category-image-variants.ts: 640 × 480) as a PNG, which its
 * bucket accepts (it refuses JPEG).
 */

const CATEGORY_BUCKET = "category-assets";
const LANDSCAPE = { cover: [1600, 1200], mid: [960, 720], card: [480, 360], thumb: [160, 120] };
const PORTRAIT = { cover: [1200, 1600], mid: [720, 960], card: [360, 480], thumb: [120, 160] };
type Sizes = typeof LANDSCAPE;

interface Answer {
  status: number;
  type: string;
  bytes: string;
}

interface Screen {
  name: string;
  path: string;
  /** Each selector must match a visible element whose picture has loaded. */
  ready: string[];
}

test.describe("INVESTIGATION — photos and category pictures by width", () => {
  const fixtures = adPageFixtures();
  const photoKeys: string[] = [];
  const categoryKeys: string[] = [];

  test.afterEach(async () => {
    await fixtures.cleanup();
    const supabase = adminClient();
    if (photoKeys.length > 0) {
      const removed = await supabase.storage.from(BUCKET).remove(photoKeys.splice(0));
      if (removed.error) throw new Error(`[e2e:iph] photo reap failed: ${removed.error.message}`);
    }
    if (categoryKeys.length > 0) {
      const removed = await supabase.storage.from(CATEGORY_BUCKET).remove(categoryKeys.splice(0));
      if (removed.error) {
        throw new Error(`[e2e:iph] category picture reap failed: ${removed.error.message}`);
      }
    }
  });

  /** A category picture's shape: a white ground, the subject centred at 85 % of the height. */
  function madePng(width: number, height: number): Buffer {
    const png = new PNG({ width, height });
    const side = Math.round(height * 0.85);
    const left = Math.round((width - side) / 2);
    const top = Math.round((height - side) / 2);
    for (let y = 0; y < height; y += 1) {
      for (let x = 0; x < width; x += 1) {
        const at = (y * width + x) * 4;
        const inside = x >= left && x < left + side && y >= top && y < top + side;
        png.data[at] = inside ? Math.round(((x - left) / side) * 255) : 255;
        png.data[at + 1] = inside ? 96 : 255;
        png.data[at + 2] = inside ? Math.round(((y - top) / side) * 255) : 255;
        png.data[at + 3] = 255;
      }
    }
    return PNG.sync.write(png);
  }

  async function upload(
    bucket: string,
    keys: string[],
    width: number,
    height: number,
    kind: "jpg" | "png" = "jpg",
  ) {
    const key = `e2e-${RUN}/iph-${rand()}-${width}x${height}.${kind}`;
    const body = kind === "png" ? madePng(width, height) : madeJpeg(width, height);
    const uploaded = await adminClient()
      .storage.from(bucket)
      .upload(key, body, { contentType: kind === "png" ? "image/png" : "image/jpeg" });
    if (uploaded.error) throw new Error(`[e2e:iph] upload failed: ${uploaded.error.message}`);
    keys.push(key);
    return key;
  }

  async function seedPhoto(listingId: string, sizes: Sizes) {
    const paths: Record<string, { partition: string; key: string }> = {};
    for (const [variant, [width, height]] of Object.entries(sizes)) {
      paths[variant] = {
        partition: "default",
        key: await upload(BUCKET, photoKeys, width, height),
      };
    }
    const inserted = await adminClient()
      .from("listing_photos")
      .insert({
        listing_id: listingId,
        storage_path: paths["cover"]?.key ?? "",
        display_order: 0,
        exif_stripped: true,
        unscreened: false,
        width: sizes.cover[0],
        height: sizes.cover[1],
        paths,
      });
    if (inserted.error) throw new Error(`[e2e:iph] photo row failed: ${inserted.error.message}`);
  }

  async function giveCategoryPicture(categoryId: string) {
    const key = await upload(CATEGORY_BUCKET, categoryKeys, 640, 480, "png");
    const url = adminClient().storage.from(CATEGORY_BUCKET).getPublicUrl(key).data.publicUrl;
    const updated = await adminClient()
      .from("categories")
      .update({ image_url: url })
      .eq("id", categoryId);
    if (updated.error) throw new Error(`[e2e:iph] picture link failed: ${updated.error.message}`);
  }

  /** Waits, bounded, for every selector's picture; answers whether they all arrived. */
  async function waitForPictures(page: Page, selectors: string[]): Promise<boolean> {
    return page
      .waitForFunction(
        (list) =>
          list.every((selector) =>
            [...document.querySelectorAll(selector)].some((element) => {
              if (element.getClientRects().length === 0) return false;
              const img =
                element instanceof HTMLImageElement ? element : element.querySelector("img");
              return img !== null && img.complete && img.naturalWidth > 0;
            }),
          ),
        selectors,
        { timeout: 30_000 },
      )
      .then(
        () => true,
        () => false,
      );
  }

  async function measure(page: Page, answers: Map<string, Answer>) {
    await settled(page);
    const rows = await page.evaluate(() => {
      const timings = new Map(
        performance.getEntriesByType("resource").map((entry) => [entry.name, entry]),
      );
      const px = (value: number) => Math.round(value * 10) / 10;
      const named = (element: Element) =>
        element.getAttribute("data-testid") ?? element.tagName.toLowerCase();
      return [...document.querySelectorAll("img")].map((img) => {
        const box = img.getBoundingClientRect();
        const style = getComputedStyle(img);
        const frame = img.parentElement;
        const picture = img.closest("[data-picture]");
        const containers: { at: string; type: string; width: number }[] = [];
        for (let up = img.parentElement; up !== null; up = up.parentElement) {
          const type = getComputedStyle(up).containerType;
          if (type !== "normal") {
            containers.push({ at: named(up), type, width: px(up.getBoundingClientRect().width) });
          }
        }
        let owner = "";
        for (let up: Element | null = img; up !== null && owner === ""; up = up.parentElement) {
          owner = up.getAttribute("data-testid") ?? "";
        }
        const source = img.currentSrc === "" ? img.src : img.currentSrc;
        const timing = timings.get(source);
        const url = new URL(source, location.href);
        return {
          owner,
          src: `${url.origin}${url.pathname}`,
          source,
          shown: img.getClientRects().length > 0,
          complete: img.complete,
          natural: `${img.naturalWidth}x${img.naturalHeight}`,
          drawn: `${px(box.width)}x${px(box.height)}`,
          drawnAt: `${px(box.x)},${px(box.y)}`,
          objectFit: style.objectFit,
          cssSize: `${style.width} / ${style.height}`,
          cssMax: `${style.maxWidth} / ${style.maxHeight}`,
          frameWidth: frame === null ? null : px(frame.getBoundingClientRect().width),
          frameHeight: frame === null ? null : px(frame.getBoundingClientRect().height),
          frameStyle: frame?.getAttribute("style") ?? "",
          frameContainer: frame === null ? "" : getComputedStyle(frame).containerType,
          picture: picture?.getAttribute("data-picture") ?? "",
          pictureWidth: picture === null ? null : px(picture.getBoundingClientRect().width),
          pictureHeight: picture === null ? null : px(picture.getBoundingClientRect().height),
          pictureContainer: picture === null ? "" : getComputedStyle(picture).containerType,
          containers,
          timingStatus:
            timing !== undefined && "responseStatus" in timing ? timing.responseStatus : null,
          timingBytes:
            timing !== undefined && "encodedBodySize" in timing ? timing.encodedBodySize : null,
        };
      });
    });
    return rows.map(({ source, ...row }) => ({ ...row, answer: answers.get(source) ?? null }));
  }

  test("IPH-1 the pictures on My listings, a My-listing page, three ad pages and the home feed, measured at each width (no assertion)", async ({
    page,
    baseURL,
  }, testInfo) => {
    // Six screens at up to two widths, each waiting up to 30 s for its pictures.
    test.setTimeout(420_000);
    const seller = await fixtures.scratchSeller({ served: true });
    const landscape = await fixtures.seed(seller, "active");
    const portrait = await fixtures.seed(seller, "active");
    const bare = await fixtures.seed(seller, "active");
    await seedPhoto(landscape.id, LANDSCAPE);
    await seedPhoto(portrait.id, PORTRAIT);
    await giveCategoryPicture(seller.parentId);
    const ids = { landscape: landscape.id, portrait: portrait.id, bare: bare.id };
    const publicIds = {
      landscape: await fixtures.publicIdOf(landscape.id),
      portrait: await fixtures.publicIdOf(portrait.id),
      bare: await fixtures.publicIdOf(bare.id),
    };

    const answers = new Map<string, Answer>();
    page.on("response", (response) => {
      if (response.request().resourceType() !== "image") return;
      const headers = response.headers();
      answers.set(response.url(), {
        status: response.status(),
        type: headers["content-type"] ?? "",
        bytes: headers["content-length"] ?? "",
      });
    });

    await page
      .context()
      .addCookies([{ name: "ethio_area", value: `ET:${seller.cityId}`, url: baseURL ?? "" }]);
    await signInViaSession(page, seller.email, seller.password);

    const routes: Record<string, unknown>[] = [];
    page.on("response", async (response) => {
      const url = new URL(response.url());
      if (!url.pathname.startsWith("/api/")) return;
      const entry: Record<string, unknown> = {
        path: url.pathname,
        method: response.request().method(),
        status: response.status(),
      };
      if (url.pathname === "/api/media/photo-links") {
        const sent: unknown = response.request().postDataJSON();
        entry["asked"] = sent;
        const body: unknown = await response.json().then(
          (value: unknown) => value,
          () => null,
        );
        const links =
          body !== null && typeof body === "object" && "links" in body ? body.links : null;
        entry["variants"] =
          links !== null && typeof links === "object"
            ? Object.fromEntries(
                Object.entries(links).map(([id, sizes]) => [
                  id,
                  sizes !== null && typeof sizes === "object" ? Object.keys(sizes) : [],
                ]),
              )
            : null;
      }
      routes.push(entry);
    });

    const listPicture = (id: string, kind: string) =>
      `[data-testid="my-listing-${id}-picture"] [data-picture="${kind}"]`;
    const card = (id: string, kind: string) =>
      `[data-testid="listing-card"][data-listing="${id}"] [data-testid="listing-card-picture"][data-picture="${kind}"]`;
    const screens: Screen[] = [
      {
        name: "my-listings",
        path: "/my-listings",
        ready: [
          listPicture(ids.landscape, "photo"),
          listPicture(ids.portrait, "photo"),
          listPicture(ids.bare, "category"),
        ],
      },
      {
        name: "my-listing-portrait",
        path: `/my-listings/${ids.portrait}`,
        ready: ['[data-testid="gallery-main"]'],
      },
      {
        name: "ad-landscape",
        path: `/l/${publicIds.landscape}`,
        ready: ['[data-testid="gallery-main"]'],
      },
      {
        name: "ad-portrait",
        path: `/l/${publicIds.portrait}`,
        ready: ['[data-testid="gallery-main"]'],
      },
      { name: "ad-no-photo", path: `/l/${publicIds.bare}`, ready: ['[data-picture="category"]'] },
      {
        name: "home-feed",
        path: "/",
        ready: [
          card(ids.landscape, "photo"),
          card(ids.portrait, "photo"),
          card(ids.bare, "category"),
        ],
      },
    ];

    const widths = testInfo.project.name === "desktop-1280" ? [1280, 1024] : [360];
    const record: Record<string, unknown>[] = [];
    for (const width of widths) {
      await page.setViewportSize({ width, height: width === 360 ? 740 : 800 });
      for (const screen of screens) {
        routes.length = 0;
        await gotoReady(page, screen.path);
        const ready = await waitForPictures(page, screen.ready);
        await page.evaluate((selector) => {
          const target = [...document.querySelectorAll(selector)].find(
            (element) => element.getClientRects().length > 0,
          );
          target?.scrollIntoView({ block: "center" });
        }, screen.ready[0] ?? "img");
        const images = await measure(page, answers);
        const label = `${screen.name}-${width}`;
        await page.screenshot({ path: testInfo.outputPath(`${label}.png`) });
        record.push({
          screen: screen.name,
          width,
          path: screen.path,
          ready,
          images,
          routes: [...routes],
        });
      }
    }

    const body = JSON.stringify({ project: testInfo.project.name, ids, record }, null, 2);
    writeFileSync(testInfo.outputPath("measurements.json"), body);
    await testInfo.attach("measurements", { body, contentType: "application/json" });
  });
});
